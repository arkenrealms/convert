cmake_minimum_required(VERSION 3.25)
project(ArkenConvert LANGUAGES C CXX)
enable_testing()
set(ARKEN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Explicit source root for monorepo qualification")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF)
set(ARKEN_SCHEMA_CPP_V1_RUNTIME_ONLY OFF)
set(ARKEN_IMAGE_CODECS_V1_RASTER_TESTS ON)
# Development qualification of the HEIC decoder (handoff: native wrapper before any admission).
# The decoder is reachable only through DecodeImage(Format::heif); no MCP tool admits HEIF yet.
set(ARKEN_IMAGE_CODECS_V1_HEIF ON)
include("${ARKEN_SOURCE_ROOT}/capsule/cmake/arken_native_defaults.cmake")
include("${ARKEN_SOURCE_ROOT}/sdk/arken_sdk_dev/cmake/ArkenPinnedNlohmannJson.cmake")
arken_configure_pinned_nlohmann_json()
set(ARKEN_NLOHMANN_TARGET arken_nlohmann_json_header_only)
set(ARKEN_NATIVE_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
include("${ARKEN_SOURCE_ROOT}/sdk/arken_sdk_dev/cmake/ArkenModuleGraph.cmake")
arken_resolve_enabled_capsules(modules ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/lib.manifest.json"
 CAPSULE_ROOT "${ARKEN_SOURCE_ROOT}/capsule" LIB_ROOT "${ARKEN_SOURCE_ROOT}/lib")
foreach(module IN LISTS modules)
 if(module MATCHES "^arken_paid_|cloudflare|^arken_x402_|^arken_codecast_|^arken_file_conversion_service_|^arken_canvas_|^arken_clang_browser_|^arken_live_machine_" OR
    (module MATCHES "^arken_mcp_" AND NOT module STREQUAL "arken_mcp_protocol_v1"))
  message(FATAL_ERROR "Capability-only Convert cannot depend on ${module}")
 endif()
endforeach()
arken_get_resolved_module_manifests(manifests)
include("${ARKEN_SOURCE_ROOT}/sdk/arken_sdk_dev/cmake/ArkenExternalPackageGraph.cmake")
arken_resolve_external_packages(ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/lib.manifest.json"
 MODULE_MANIFESTS ${manifests} WORKSPACE_ROOT "${ARKEN_SOURCE_ROOT}")
arken_add_resolved_module_targets(BINARY_ROOT "${CMAKE_CURRENT_BINARY_DIR}/capsules")
set_target_properties(arken-convert-commands PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
arken_add_mcp_stdio_checks()
arken_add_native_files_checks()
include("${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_v1/cmake/ArkenFileConversionImages.cmake")
arken_add_file_conversion_image_tests()
add_subdirectory("${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_commands_v1/tests" "command-tests")
