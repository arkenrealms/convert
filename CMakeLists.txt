cmake_minimum_required(VERSION 3.25)
project(ArkenConvertMcp LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF)
option(ARKEN_CONVERT_MCP_CHECKS "Build native service and image/payment checks" ON)
if(ARKEN_CONVERT_MCP_CHECKS AND NOT ARKEN_BROWSER_CPP)
    enable_testing()
    set(ARKEN_IMAGE_CODECS_V1_RASTER_TESTS ON)
endif()
set(ARKEN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Explicit Convert MCP source root")
set(_sdk "${ARKEN_SOURCE_ROOT}/sdk/arken_sdk_dev")
include("${ARKEN_SOURCE_ROOT}/capsule/cmake/arken_native_defaults.cmake")
include("${_sdk}/cmake/ArkenPinnedNlohmannJson.cmake")
arken_configure_pinned_nlohmann_json()
set(ARKEN_NLOHMANN_TARGET arken_nlohmann_json_header_only)
set(ARKEN_NATIVE_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${ARKEN_NATIVE_GENERATED_DIR}")
include("${_sdk}/cmake/ArkenModuleGraph.cmake")
set(root_manifest "${CMAKE_CURRENT_SOURCE_DIR}/lib.manifest.json")
if(ARKEN_CONVERT_MCP_CHECKS AND NOT ARKEN_BROWSER_CPP)
    set(root_manifest "${CMAKE_CURRENT_SOURCE_DIR}/qualification.manifest.json")
endif()
arken_resolve_enabled_capsules(modules ROOT_MANIFEST "${root_manifest}"
    CAPSULE_ROOT "${ARKEN_SOURCE_ROOT}/capsule" LIB_ROOT "${ARKEN_SOURCE_ROOT}/lib")
arken_get_resolved_module_manifests(manifests)
include("${_sdk}/cmake/ArkenExternalPackageGraph.cmake")
arken_resolve_external_packages(ROOT_MANIFEST "${root_manifest}"
    MODULE_MANIFESTS ${manifests} WORKSPACE_ROOT "${ARKEN_SOURCE_ROOT}")
arken_add_resolved_module_targets(BINARY_ROOT "${CMAKE_CURRENT_BINARY_DIR}/capsules")
foreach(forbidden arken_file_conversion_v1 arken_convert_collection_v1 arken_canvas_operation_kernel_v1
        arken_canvas_execution_v1 arken_player_v1 arken_pdf_v1 arken_mcp_suggestions_v1
        arken_clang_browser_v1 arken_mcp_protocol_native_v1 arken_cloudflare_client_v1
        arken_paid_mcp_cloudflare_v1 arken-conversion-cloudflare)
    if(TARGET ${forbidden})
        message(FATAL_ERROR "Convert MCP must not configure unrelated target ${forbidden}")
    endif()
endforeach()
add_library(arken::convert_mcp ALIAS arken_file_conversion_service_v1)
if(ARKEN_CONVERT_MCP_CHECKS AND NOT ARKEN_BROWSER_CPP)
    include("${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_service_v1/cmake/PaymentChecks.cmake")
    arken_add_conversion_payment_checks()
    include("${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_service_v1/cmake/OfflinePaymentChecks.cmake")
    arken_add_conversion_offline_payment_checks()
    arken_add_source_package_checks()
    add_executable(arken-convert-source-package "${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_service_v1/tools/native/source_package_main.cpp")
    target_link_libraries(arken-convert-source-package PRIVATE arken_paid_mcp_source_package_v1)
    add_test(NAME ConvertMcp.SourcePackageVersion COMMAND arken-convert-source-package --invalid --version /not/a/path)
    set_tests_properties(ConvertMcp.SourcePackageVersion PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "^arken-convert-source-package-r6250")
    include("${ARKEN_SOURCE_ROOT}/capsule/arken_file_conversion_v1/cmake/ArkenFileConversionImages.cmake")
    arken_add_file_conversion_image_tests()
    arken_add_browser_image_tests()
    add_executable(arken-convert-mcp-contract-tests contract_checks.cpp)
    target_link_libraries(arken-convert-mcp-contract-tests PRIVATE arken::convert_mcp arken_nlohmann_json_header_only)
    add_test(NAME ConvertMcp.Contracts COMMAND arken-convert-mcp-contract-tests)
    add_test(NAME ConvertMcp.Version COMMAND arken-convert-mcp-contract-tests --invalid --version)
    set_tests_properties(ConvertMcp.Contracts ConvertMcp.Version PROPERTIES TIMEOUT 5)
    set_tests_properties(ConvertMcp.Version PROPERTIES PASS_REGULAR_EXPRESSION "arken-convert-mcp-contract-tests-r6242")
endif()
message(STATUS "Convert MCP owners: ${modules}")
