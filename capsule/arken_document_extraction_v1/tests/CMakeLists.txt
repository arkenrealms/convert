cmake_minimum_required(VERSION 3.25)
project(ArkenDocumentExtractionTests LANGUAGES C CXX)
get_filename_component(workspace "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenExternalPackageGraph.cmake")
arken_resolve_external_packages(ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/lib.manifest.json" WORKSPACE_ROOT "${workspace}")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." document_extraction)
add_executable(arken-document-extraction-tests extraction_tests.cpp)
target_compile_features(arken-document-extraction-tests PRIVATE cxx_std_20)
target_link_libraries(arken-document-extraction-tests PRIVATE arken::document_extraction_v1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(arken-document-extraction-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME DocumentExtraction.Version COMMAND arken-document-extraction-tests --version --invalid)
set_tests_properties(DocumentExtraction.Version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "arken-document-extraction-v1-r1")
foreach(group zip xml docx rtf markdown limits determinism mutation)
  add_test(NAME DocumentExtraction.${group} COMMAND arken-document-extraction-tests ${group})
  set_tests_properties(DocumentExtraction.${group} PROPERTIES TIMEOUT 60)
endforeach()
# Tabular conversion (tabular.hpp): CSV/TSV/JSON, XLSX and ODS.
add_executable(arken-document-tabular-tests tabular_tests.cpp)
target_compile_features(arken-document-tabular-tests PRIVATE cxx_std_20)
target_link_libraries(arken-document-tabular-tests PRIVATE arken::document_extraction_v1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(arken-document-tabular-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME DocumentTabular.Version COMMAND arken-document-tabular-tests --version --invalid)
set_tests_properties(DocumentTabular.Version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "arken-document-tabular-v1-r1")
foreach(group csv json xlsx ods limits determinism mutation)
  add_test(NAME DocumentTabular.${group} COMMAND arken-document-tabular-tests ${group})
  set_tests_properties(DocumentTabular.${group} PROPERTIES TIMEOUT 120)
endforeach()
