cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ArkenMcpEconomics LANGUAGES C CXX)
 get_filename_component(workspace "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
 include("${workspace}/capsule/cmake/arken_native_defaults.cmake")
 include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenPinnedNlohmannJson.cmake")
 arken_configure_pinned_nlohmann_json()
 set(ARKEN_NATIVE_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
 set(ARKEN_SCHEMA_CPP_V1_RUNTIME_ONLY ON CACHE BOOL "Committed schema" FORCE)
 set(BUILD_TESTING OFF)
 include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenModuleGraph.cmake")
 arken_resolve_enabled_capsules(modules ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/capsule.manifest.json" CAPSULE_ROOT "${workspace}/capsule" LIB_ROOT "${workspace}/lib")
 foreach(module IN LISTS modules)
  if(NOT module STREQUAL "arken_mcp_economics_v1")
   add_subdirectory("${workspace}/capsule/${module}" "capsules/${module}")
  endif()
 endforeach()
endif()
add_library(arken_mcp_economics_v1 STATIC src/economics.cpp src/pricing.cpp)
target_compile_features(arken_mcp_economics_v1 PUBLIC cxx_std_20)
target_include_directories(arken_mcp_economics_v1 PUBLIC "${CMAKE_CURRENT_LIST_DIR}/..")
target_compile_options(arken_mcp_economics_v1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(arken_mcp_economics_protocol_v1 STATIC src/budget_protocol.cpp)
target_compile_features(arken_mcp_economics_protocol_v1 PUBLIC cxx_std_20)
target_link_libraries(arken_mcp_economics_protocol_v1 PUBLIC arken_mcp_economics_v1 PRIVATE arken_schema_cpp_v1_tool_json arken_digest_v1)
target_compile_options(arken_mcp_economics_protocol_v1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ARKEN_BROWSER_CPP)
 include("${CMAKE_CURRENT_LIST_DIR}/../../sdk/arken_sdk_dev/cmake/ArkenBrowserCpp.cmake")
 add_executable(arken-mcp-economics-module platform/wasm/bridge.cpp)
 target_link_libraries(arken-mcp-economics-module PRIVATE arken_mcp_economics_protocol_v1)
 arken_browser_cpp_reactor(arken-mcp-economics-module NO_HOST_IO MAX_MEMORY_BYTES 8388608
  EXPORT_FUNCTIONS arken_budget_prepare arken_budget_run arken_budget_result arken_budget_clear)
elseif(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 enable_testing()
 include("${CMAKE_CURRENT_LIST_DIR}/cmake/PolicyChecks.cmake")
 arken_add_mcp_economics_checks()
endif()
