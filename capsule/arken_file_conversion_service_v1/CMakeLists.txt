cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ArkenFileConversionService LANGUAGES C CXX)
    include(CTest)
    get_filename_component(workspace "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    include("${workspace}/capsule/cmake/arken_native_defaults.cmake")
    set(ARKEN_NATIVE_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
    file(MAKE_DIRECTORY "${ARKEN_NATIVE_GENERATED_DIR}")
    set(ARKEN_SCHEMA_CPP_V1_RUNTIME_ONLY ON CACHE BOOL "Use committed generated conversion contracts" FORCE)
    include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenPinnedNlohmannJson.cmake")
    arken_configure_pinned_nlohmann_json()
    set(ARKEN_NLOHMANN_TARGET arken_nlohmann_json_header_only)
    include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenModuleGraph.cmake")
    arken_resolve_enabled_capsules(modules ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/capsule.manifest.json"
        CAPSULE_ROOT "${workspace}/capsule" LIB_ROOT "${workspace}/lib")
    set(manifests)
    foreach(module IN LISTS modules)
        list(APPEND manifests "${workspace}/capsule/${module}/capsule.manifest.json")
    endforeach()
    include("${workspace}/sdk/arken_sdk_dev/cmake/ArkenExternalPackageGraph.cmake")
    arken_resolve_external_packages(ROOT_MANIFEST "${CMAKE_CURRENT_LIST_DIR}/capsule.manifest.json"
        MODULE_MANIFESTS ${manifests} WORKSPACE_ROOT "${workspace}")
    foreach(module IN LISTS modules)
        if(NOT module STREQUAL "arken_file_conversion_service_v1")
            add_subdirectory("${workspace}/capsule/${module}" "capsules/${module}")
        endif()
    endforeach()
endif()
add_library(arken_file_conversion_service_v1 STATIC src/commands.cpp src/payment.cpp src/payment_recovery.cpp src/payment_packet.cpp src/x402.cpp src/paid_conversion.cpp src/settlement_reconciliation.cpp src/public_admission.cpp src/provider_authorization.cpp)
target_compile_features(arken_file_conversion_service_v1 PUBLIC cxx_std_20)
target_include_directories(arken_file_conversion_service_v1 PUBLIC "${CMAKE_CURRENT_LIST_DIR}/.."
    PRIVATE "${CMAKE_CURRENT_LIST_DIR}/generated")
target_link_libraries(arken_file_conversion_service_v1 PUBLIC arken_file_conversion_v1
    PRIVATE arken_schema_cpp_v1_tool_json arken_digest_v1)
if(ARKEN_BROWSER_CPP)
    include("${CMAKE_CURRENT_LIST_DIR}/../arken_file_conversion_v1/cmake/ArkenFileConversionReactor.cmake")
    arken_file_conversion_reactor(arken-file-conversion-service-module)
    target_sources(arken-file-conversion-service-module PRIVATE platform/wasm/bridge.cpp)
    target_link_libraries(arken-file-conversion-service-module PRIVATE arken_file_conversion_service_v1)
endif()
