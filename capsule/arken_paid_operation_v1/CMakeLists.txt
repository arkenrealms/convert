cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ArkenPaidOperation LANGUAGES CXX)
 enable_testing()
endif()
add_library(arken_paid_operation_v1 INTERFACE)
target_compile_features(arken_paid_operation_v1 INTERFACE cxx_std_20)
target_include_directories(arken_paid_operation_v1 INTERFACE "${CMAKE_CURRENT_LIST_DIR}/..")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 add_executable(arken-paid-operation-tests tests/policy.cpp)
 target_link_libraries(arken-paid-operation-tests PRIVATE arken_paid_operation_v1)
 target_compile_options(arken-paid-operation-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME PaidOperation.Contracts COMMAND arken-paid-operation-tests)
 add_test(NAME PaidOperation.Version COMMAND arken-paid-operation-tests --invalid --version)
 set_tests_properties(PaidOperation.Version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "r4511")
endif()
