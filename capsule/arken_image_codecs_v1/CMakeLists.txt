cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ArkenImageCodecs LANGUAGES CXX)
    include(CTest)
    include(${CMAKE_CURRENT_SOURCE_DIR}/../../sdk/arken_sdk_dev/cmake/ArkenExternalPackageGraph.cmake)
    arken_resolve_external_packages(ROOT_MANIFEST ${CMAKE_CURRENT_SOURCE_DIR}/capsule.manifest.json)
endif()
add_library(arken_image_codecs_v1_test_support INTERFACE)
add_library(arken::image_codecs_v1_test_support ALIAS arken_image_codecs_v1_test_support)
target_include_directories(arken_image_codecs_v1_test_support INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/..)
option(ARKEN_IMAGE_CODECS_V1_RASTER_TESTS "Build focused raster codec tests" OFF)
if(BUILD_TESTING OR ARKEN_IMAGE_CODECS_V1_RASTER_TESTS)
    add_executable(arken-image-encode-tests tests/encode_tests.cpp)
    target_link_libraries(arken-image-encode-tests PRIVATE arken::image_codecs_v1)
    add_test(NAME ArkenImageCodecsV1.Encode COMMAND arken-image-encode-tests)
    add_test(NAME ArkenImageCodecsV1.EncodeVersion COMMAND arken-image-encode-tests --version --invalid)
    set_tests_properties(ArkenImageCodecsV1.EncodeVersion PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "r4305")
    add_executable(arken-raster-codec-tests tests/raster_tests.cpp)
    target_link_libraries(arken-raster-codec-tests PRIVATE arken::image_codecs_v1 arken::external_stb_image_v1)
    add_test(NAME ArkenImageCodecsV1.Raster COMMAND arken-raster-codec-tests)
    add_test(NAME ArkenImageCodecsV1.RasterVersion COMMAND arken-raster-codec-tests --version --invalid)
    set_tests_properties(ArkenImageCodecsV1.RasterVersion PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "r4266")
endif()
if(BUILD_TESTING)
    add_executable(arken-image-texture-tests tests/texture_tests.cpp)
    target_link_libraries(arken-image-texture-tests PRIVATE arken::image_codecs_v1)
    add_test(NAME ArkenImageCodecsV1.Texture COMMAND arken-image-texture-tests)
    add_test(NAME ArkenImageCodecsV1.TextureVersion COMMAND arken-image-texture-tests --version)
    set_tests_properties(ArkenImageCodecsV1.TextureVersion PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "r4260")
    add_executable(arken-image-codecs-tests tests/codec_tests.cpp)
    target_link_libraries(arken-image-codecs-tests PRIVATE arken::image_codecs_v1 arken::image_codecs_v1_test_support)
    add_test(NAME ArkenImageCodecsV1.Decode COMMAND arken-image-codecs-tests)
    add_test(NAME ArkenImageCodecsV1.Version COMMAND arken-image-codecs-tests --version)
    set_tests_properties(ArkenImageCodecsV1.Version PROPERTIES TIMEOUT 5 PASS_REGULAR_EXPRESSION "r4222")
endif()
add_library(arken_image_codecs_v1 STATIC src/codec.cpp src/pcx.cpp src/cel.cpp src/dc6.cpp src/blp.cpp src/blp2.cpp src/block_decode.cpp src/dds.cpp src/tga.cpp src/raster.cpp src/raster_encode.cpp)
add_library(arken::image_codecs_v1 ALIAS arken_image_codecs_v1)
target_compile_features(arken_image_codecs_v1 PUBLIC cxx_std_20)
target_include_directories(arken_image_codecs_v1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(arken_image_codecs_v1 PRIVATE arken::external_stb_image_v1)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(arken_image_codecs_v1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
